    %% Almacenes de Datos

        D14_Cat[(Categoria)]

    end

    P6_3_1 --> C_Rol{¿Rol del JWT<br>es ADMINISTRADOR?}
    C_Rol -->|No| Err_Forbidden([Error 403: Prohibido. Requiere rol Administrador])

    %% los eventos vinculados conservan su categoria, sin registros huerfanos.

    C_Ultima -->|Sí| Err_Ultima([Error 409: No se puede desactivar la última categoría activa])
    C_Ultima -->|No| P6_3_4C[6.3.4C: Actualizar activa = true/false]
    P6_3_4C -->|Ejecutar UPDATE| D14_Cat
    %% Efecto: deja de ofrecerse para eventos nuevos y en los filtros del catálogo

    Err_Forbidden --> FIN
    Err_Duplicado --> FIN
